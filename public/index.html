function verifierMotDePasse() { var saisie = document.getElementById('profPasswordInput').value; var errorMsg = document.getElementById('authErrorMsg'); fetch('/api/connexion', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ password: saisie }) }) .then(response => response.json()) .then(data => { if (data.success) { document.getElementById('authOverlay').style.display = 'none'; } else { errorMsg.style.display = 'block'; document.getElementById('profPasswordInput').value = ''; document.getElementById('profPasswordInput').focus(); } }) .catch(err => { console.error(err); errorMsg.textContent = "Erreur de connexion au serveur."; errorMsg.style.display = 'block'; }); }
